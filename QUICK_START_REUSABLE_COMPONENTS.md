# Quick Start - Using New Reusable Components 🚀

## What Was Created

Three new production-ready components:

### 1. **GenericRuleEditorComponent** (`app-generic-rule-editor`)
- Standalone logical rule editor
- Full FormControl support
- All features (autocomplete, validation, warnings, help, examples)
- Config-driven

### 2. **DroolsEditorComponent** (`app-drools-editor`)
- Standalone Drools viewer/editor
- Full FormControl support
- Auto-generate or manual edit modes
- Copy to clipboard

### 3. **SideBySideContainerComponent** (`app-side-by-side-container`)
- Composes both editors
- FormGroup integration
- Theme toggle, save button, clear all
- Horizontal or vertical layout

## Quick Usage

### Option 1: Use Side-by-Side Container (Easiest)

```typescript
import { Component } from '@angular/core';
import { SideBySideContainerComponent } from './components/side-by-side-container/side-by-side-container.component';

@Component({
  selector: 'app-root',
  standalone: true,
  imports: [SideBySideContainerComponent],
  template: `
    <app-side-by-side-container
      (saved)="onSave($event)"
    ></app-side-by-side-container>
  `
})
export class AppComponent {
  onSave(data: { logicalRule: string; drools: string }) {
    console.log('Saved:', data);
  }
}
```

### Option 2: Use with Reactive Forms

```typescript
import { Component } from '@angular/core';
import { FormGroup, FormControl, ReactiveFormsModule } from '@angular/forms';
import { SideBySideContainerComponent } from './components/side-by-side-container/side-by-side-container.component';

@Component({
  standalone: true,
  imports: [ReactiveFormsModule, SideBySideContainerComponent],
  template: `
    <form [formGroup]="myForm" (ngSubmit)="onSubmit()">
      <app-side-by-side-container
        [formGroup]="myForm"
        logicalRuleControlName="rule"
        droolsControlName="drools"
      ></app-side-by-side-container>
      
      <button type="submit">Submit</button>
    </form>
  `
})
export class AppComponent {
  myForm = new FormGroup({
    rule: new FormControl(''),
    drools: new FormControl('')
  });
  
  onSubmit() {
    console.log(this.myForm.value);
  }
}
```

### Option 3: Use Independently

```typescript
import { Component } from '@angular/core';
import { FormControl, ReactiveFormsModule } from '@angular/forms';
import { GenericRuleEditorComponent } from './components/generic-rule-editor/generic-rule-editor.component';
import { DroolsEditorComponent } from './components/drools-editor/drools-editor.component';

@Component({
  standalone: true,
  imports: [ReactiveFormsModule, GenericRuleEditorComponent, DroolsEditorComponent],
  template: `
    <h3>Logical Rule:</h3>
    <app-generic-rule-editor
      [formControl]="ruleControl"
      [minHeight]="'300px'"
    ></app-generic-rule-editor>
    
    <h3>Drools Expression:</h3>
    <app-drools-editor
      [formControl]="droolsControl"
      [minHeight]="'300px'"
    ></app-drools-editor>
  `
})
export class AppComponent {
  ruleControl = new FormControl('');
  droolsControl = new FormControl('');
}
```

## Configuration Options

### GenericRuleEditor

```typescript
<app-generic-rule-editor
  [formControl]="control"
  [showExamples]="true"          // Show examples button
  [showHelp]="true"              // Show help button
  [showTokenPreview]="true"      // Show token preview
  [showWarnings]="true"          // Show validation warnings
  [minHeight]="'200px'"          // Min height
  placeholder="Custom placeholder..."
  (ruleChanged)="onChange($event)"
  (validationChanged)="onValidation($event)"
></app-generic-rule-editor>
```

### DroolsEditor

```typescript
<app-drools-editor
  [formControl]="control"
  [readonly]="false"             // Editable or readonly
  [autoGenerate]="true"          // Auto-generate on focus
  [showCopyButton]="true"        // Show copy button
  [showHint]="true"              // Show hint text
  [minHeight]="'200px'"
  label="My Drools Expression"
  (generateRequested)="onGenerate()"
></app-drools-editor>
```

### SideBySideContainer

```typescript
<app-side-by-side-container
  title="Custom Title"
  [showThemeToggle]="true"       // Show theme toggle
  [showAdvancedMode]="true"      // Show advanced mode
  [showSaveButton]="true"        // Show save button
  [autoGenerateDrools]="true"    // Auto-generate on focus
  [layout]="'horizontal'"        // or 'vertical'
  [formGroup]="myForm"           // Optional external form
  (saved)="onSave($event)"
  (themeChanged)="onTheme($event)"
></app-side-by-side-container>
```
